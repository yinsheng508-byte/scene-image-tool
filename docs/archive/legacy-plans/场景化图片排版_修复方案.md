## 一、问题概述

### 1.1 核心问题
1. **性能问题**: 上传多张图片后切换时界面卡顿
2. **用户体验问题**: 每次下载都需要手动选择保存位置
3. **下载逻辑问题**:
   - 单张图片应直接下载为图片文件,而非压缩包
   - 多张图片应保存到文件夹,而非压缩包
   - 文件夹命名需包含图片数量和下载时间
4. **缺少状态反馈**: 下载成功或失败时没有明确提示

### 1.2 影响范围
- 文件: `desktop/main.js`, `desktop/preload.js`, `desktop/renderer/compose.js`, `desktop/renderer/compose.html`, `desktop/renderer/styles.css`

### 2.1 性能优化 (解决卡顿)

#### 问题分析
**根本原因**:
- 在 `compose.js` 的 `renderWebGL()` 函数中,每一帧渲染都执行大量 `console.log`

- 日志位置:
  - `renderWebGL()`: 第327-400行
  - `updateWebGLPptTexture()`: 第274-291行
  - `updateWebGLBgTexture()`: 第294-313行
  - `downloadAllBtn` 事件处理: 第902-1190行

```javascript
// 创建目录(带自动重命名)
ipcMain.handle("file:createDirectory", async (event, payload) => {
  const { baseDir, dirName } = payload;
  if (!baseDir || !dirName) {
    return { ok: false, error: "参数不完整" };
  }
  try {
    let fullPath = path.join(baseDir, dirName);

    // 检查是否存在,如果存在则自动添加序号
    if (fs.existsSync(fullPath)) {
      let counter = 1;
      while (fs.existsSync(path.join(baseDir, `${dirName}_${counter}`))) {
        counter++;
        if (counter > 999) { // 防止无限循环
          return { ok: false, error: "无法创建目录,已存在过多同名文件夹" };
        }
      }
      fullPath = path.join(baseDir, `${dirName}_${counter}`);
    }

    await fs.promises.mkdir(fullPath, { recursive: true });
    return { ok: true, path: fullPath };
  } catch (error) {
    return { ok: false, error: error.message };
  }
});

// 保存图片文件
ipcMain.handle("file:saveImage", async (event, payload) => {
  const { directory, fileName, buffer } = payload;
  if (!directory || !fileName || !buffer) {
    return { ok: false, error: "参数不完整" };
  }
  try {
    const fullPath = path.join(directory, fileName);
    // buffer 可能是 ArrayBuffer 或 Uint8Array
    const bufferData = Buffer.from(buffer);
    await fs.promises.writeFile(fullPath, bufferData);
    return { ok: true, path: fullPath };
  } catch (error) {
    return { ok: false, error: error.message };
  }
});
```

```javascript
// 格式化时间戳
function getTimestamp() {
  const now = new Date();
  const yyyy = now.getFullYear();
  const MM = String(now.getMonth() + 1).padStart(2, '0');
  const dd = String(now.getDate()).padStart(2, '0');
  const HH = String(now.getHours()).padStart(2, '0');
  const mm = String(now.getMinutes()).padStart(2, '0');
  const ss = String(now.getSeconds()).padStart(2, '0');
  return `${yyyy}${MM}${dd}_${HH}${mm}${ss}`;
}

// 显示Toast提示(居中显示)
function showToast(message, type = 'info', duration = 3000) {
  const toast = document.getElementById('composeToast');
  const content = document.getElementById('composeToastContent');
  if (!toast || !content) return;

  // 设置背景颜色
  let bgColor = 'rgba(0, 0, 0, 0.85)';
  if (type === 'success') {
    bgColor = 'rgba(16, 185, 129, 0.9)'; // 绿色
  } else if (type === 'error') {
    bgColor = 'rgba(239, 68, 68, 0.9)'; // 红色
  } else if (type === 'warning') {
    bgColor = 'rgba(245, 158, 11, 0.9)'; // 橙色
  }

  toast.style.background = bgColor;
  content.textContent = message;
  toast.style.display = 'block';

  if (duration > 0) {
    setTimeout(() => {
      toast.style.display = 'none';
    }, duration);
  }
}

// 隐藏Toast提示
function hideToast() {
  const toast = document.getElementById('composeToast');
  if (toast) toast.style.display = 'none';
}

// 生成唯一文件名(避免冲突)
function getUniqueFileName(baseName, extension, existingNames) {
  let fileName = `${baseName}.${extension}`;
  if (!existingNames.has(fileName.toLowerCase())) {
    return fileName;
  }

  let counter = 1;
  while (existingNames.has(`${baseName}_${counter}.${extension}`.toLowerCase())) {
    counter++;
  }
  return `${baseName}_${counter}.${extension}`;
}
```

```javascript
    downloadAllBtn.disabled = false;
  }
});
```
